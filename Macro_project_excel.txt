' =============================================================================
' PNL Explainer BiG - Optimized & Corrected Workflow Module (v2)
'
' This revision applies the full diagnostic remediation:
'   * Bond ticker resolver tries /isin/ first, then Corp/Govt/Mtge/M-Mkt.
'   * I-spread / G-spread / DV01 / SpreadDuration are DERIVED, not pulled from
'     fields that are Not Applicable / Invalid for this book.
'   * All config references inside FormulaR1C1 use CfgR1C1() (no literal sheet
'     name concatenation, which broke DaysToDeliv and futures price pulls).
'   * Futures CTD bond resolved from OPICS CTD_ISIN, not FUT_CTD_BOND.
'   * Futures generic ticker built from a CurveMap/FutMap lookup table, not
'     blind  code & "1 Comdty".
'   * T0 missing is preserved as "" (never silently coerced to 0), so spread
'     deltas are blanked instead of producing fake spread PnL.
'   * Status columns require T0 presence; PNL_Attribution is fully rebuilt to
'     clear stale #REF! formulas.
'
' Buttons:
'   0) SetupWorkbookFinalLayout
'   1) LoadOPICS_Bonds
'   2) LoadOPICS_Hedges
'   3) Refresh_BBG_T1_Step3   (3A curves, 3B tickers, 3C bonds, 3D futures, 3E derived)
'   4) Freeze_T0_Snapshot_Step4
'   5) RecalculatePNL_Step5
'   *) RebuildPNLOnly         (fast PNL-only rebuild to clear #REF!)
'
' Core PnL mapping:  y = r + g + q + i
'   r = OIS rate;  g = Gov - OIS;  q = Swap - Gov;  i = I-spread
' =============================================================================

' =============================================================================
' DIAGNOSTIC STATE
'   mLastFormulaTarget records the most recent range a formula was being written
'   to, so step error handlers can report exactly WHERE a .FormulaR1C1 assignment
'   failed (e.g. an over-nested formula -> Excel run-time error 1004) instead of a
'   hardcoded "most likely location" guess.
' =============================================================================

Private mLastFormulaTarget As String

' Captures the live Err object into a formatted string.  MUST be called as the very
' first statement of an error handler, BEFORE any Application.* reset (re-enabling
' events / restoring Calculation fires recalcs that clear Err -> the old handlers
' printed "Error 0").  Includes the last formula target when one is set.
Private Function CaptureErrInfo() As String
    Dim s As String
    s = "Error " & Err.Number & ": " & Err.Description
    If Len(mLastFormulaTarget) > 0 Then
        s = s & vbCrLf & "Last formula target: " & mLastFormulaTarget
    End If
    CaptureErrInfo = s
End Function

' =============================================================================
' CONFIG R1C1 HELPER  (the single fix that repairs DaysToDeliv & futures pulls)
'   Returns an absolute cross-sheet R1C1 reference to a Config cell, e.g.
'   CfgR1C1(CFG_T1_DATE) -> "'Config'!R5C2"
' =============================================================================

Private Function CfgR1C1(ByVal a1Cell As String) As String
    Dim rng As Range
    Set rng = ThisWorkbook.Worksheets(SH_CONFIG).Range(a1Cell)
    CfgR1C1 = "'" & SH_CONFIG & "'!R" & rng.Row & "C" & rng.Column
End Function

Private Sub SetupNamedRanges()

    AddOrReplaceName "Cfg_T0", "'" & SH_CONFIG & "'!" & CFG_T0_DATE
    AddOrReplaceName "Cfg_T1", "'" & SH_CONFIG & "'!" & CFG_T1_DATE
    AddOrReplaceName "Cfg_FxFix", "'" & SH_CONFIG & "'!" & CFG_FX_FIX_SOURCE
    AddOrReplaceName "Cfg_FutPxField", "'" & SH_CONFIG & "'!" & CFG_FUT_PRICE_FIELD
    AddOrReplaceName "Cfg_SpreadFwk", "'" & SH_CONFIG & "'!" & CFG_SPREAD_FRAMEWORK
End Sub

Private Sub AddOrReplaceName(ByVal nameText As String, ByVal refersToText As String)
    On Error Resume Next
    ThisWorkbook.Names(nameText).Delete
    On Error GoTo 0

    If Left$(refersToText, 1) <> "=" Then refersToText = "=" & refersToText
    ThisWorkbook.Names.Add Name:=nameText, RefersTo:=refersToText
End Sub


' =============================================================================
' FUTURES TICKER MAP SHEET  (OPICS contract code -> Bloomberg generic)
'   Replaces blind  code & "1 Comdty".  Verify the "verify" rows on the desk.
' =============================================================================

Private Sub EnsureFutMapSheet()
    Dim ws As Worksheet
    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets(SH_FUTMAP)
    On Error GoTo 0

    If ws Is Nothing Then
        Set ws = ThisWorkbook.Worksheets.Add(After:=ThisWorkbook.Worksheets(ThisWorkbook.Worksheets.Count))
        ws.Name = SH_FUTMAP
    End If

    ws.Range("A1:D1").Value = Array("OPICS_Code", "Bloomberg_Generic", "Description", "Status")
    ws.Range("A1:D1").Font.Bold = True

    ' Only (re)write seed rows if the table is empty, so manual edits survive.
    If CleanText(ws.Range("A2").Value) = "" Then
        Dim m As Variant, i As Long
        m = Array( _
            Array("DU", "DU1 Comdty", "Euro Schatz", "OK"), _
            Array("OE", "OE1 Comdty", "Euro Bobl", "OK"), _
            Array("RX", "RX1 Comdty", "Euro Bund", "OK"), _
            Array("UB", "UB1 Comdty", "Euro Buxl", "OK"), _
            Array("OAT", "OAT1 Comdty", "OAT future", "OK"), _
            Array("IK", "IK1 Comdty", "BTP future", "verify"), _
            Array("FR", "", "Mapping unverified", "verify"), _
            Array("RP", "", "Mapping unverified", "verify"), _
            Array("G", "", "Mapping unverified", "verify"), _
            Array("EC", "", "Mapping unverified", "verify"), _
            Array("UXY", "", "Mapping unverified", "verify"))
        For i = LBound(m) To UBound(m)
            ws.Cells(i + 2, 1).Resize(1, 4).Value = m(i)
        Next i
    End If

    AddOrReplaceName "FutMapTable", "'" & SH_FUTMAP & "'!$A$2:$B$200"
End Sub

' =============================================================================
' BUTTON 1 - LOAD OPICS BONDS
' =============================================================================

Public Sub LoadOPICS_Bonds()
    Dim wb As Workbook: Set wb = ThisWorkbook
    Dim wsCfg As Worksheet: Set wsCfg = wb.Worksheets(SH_CONFIG)
    Dim wsBnd As Worksheet: Set wsBnd = wb.Worksheets(SH_BONDS)

    Update_Config_Dates_5pmUTC wsCfg
    SetupExtendedConfig wsCfg
    SetupBondsFinalHeaders wsBnd

    Dim htcInvTypes As String
    Dim sellInvTypes As String
    Dim isinType As String
    Dim asOf As String
    Dim branchCode As String

    htcInvTypes = CleanText(wsCfg.Range(CFG_HTC_PORTS).Value)
    sellInvTypes = CleanText(wsCfg.Range(CFG_SELL_PORTS).Value)
    isinType = CleanText(wsCfg.Range(CFG_ISIN_TYPE).Value)
    branchCode = CleanText(wsCfg.Range(CFG_BR).Value)
    asOf = GetAsOfDateString(wsCfg)

    If htcInvTypes = "" And sellInvTypes = "" Then
        MsgBox "No INVTYPE codes in Config B8/B9 (e.g. B8=H, B9=A).", vbExclamation
        Exit Sub
    End If

    If isinType = "" Then
        MsgBox "Missing ISIN type in Config " & CFG_ISIN_TYPE & ".", vbExclamation
        Exit Sub
    End If

    If branchCode = "" Then
        MsgBox "Missing OPICS branch in Config " & CFG_BR & ".", vbExclamation
        Exit Sub
    End If

    Dim allInvTypes As String
    allInvTypes = BuildInClause(htcInvTypes) & "," & BuildInClause(sellInvTypes)
    allInvTypes = TrimChar(Replace(allInvTypes, ",,", ","), ",")

    If allInvTypes = "" Then
        MsgBox "INVTYPE list could not be built from Config B8/B9.", vbExclamation
        Exit Sub
    End If

    Dim conn As Object: Set conn = CreateObject("ADODB.Connection")
    If Not OpenOPICS(conn, wsCfg) Then Exit Sub

    Dim rs As Object: Set rs = CreateObject("ADODB.Recordset")

    On Error GoTo BondErr

    Dim sql As String
    sql = GetBondSelectQuery(allInvTypes, isinType, asOf, branchCode)

    Debug.Print sql

    rs.Open sql, conn, 0, 1

    wsBnd.Range("A" & BOND_DATA_ROW & ":CG" & CStr(BOND_DATA_ROW + N_BOND_ROWS - 1)).ClearContents

    Dim r As Long: r = BOND_DATA_ROW
    Dim bondCount As Long: bondCount = 0

    Do While Not rs.EOF And r < BOND_DATA_ROW + N_BOND_ROWS

        Dim coupon As Double: coupon = 0
        If Not IsNull(Fld(rs, "Coupon_Dec")) Then
            coupon = CDblSafe(Fld(rs, "Coupon_Dec"))
            If Abs(coupon) > 1 Then coupon = coupon / 100
        End If

        Dim fundRate As Double: fundRate = 0
        If Not IsNull(Fld(rs, "FundingRate_Dec")) Then
            fundRate = CDblSafe(Fld(rs, "FundingRate_Dec"))
            If Abs(fundRate) > 1 Then fundRate = fundRate / 100
        End If

        With wsBnd
            .Cells(r, 1).Value = NullToStr(Fld(rs, "ISIN"))
            .Cells(r, 2).Value = NullToStr(Fld(rs, "Name"))
            .Cells(r, 3).Value = NullToStr(Fld(rs, "Currency"))
            .Cells(r, 4).Value = coupon
            .Cells(r, 5).Value = NullToStr(Fld(rs, "CouponFreq"))
            .Cells(r, 6).Value = NullToDate(Fld(rs, "MaturityDate"))
            .Cells(r, 7).Value = CDblSafe(Fld(rs, "Notional"))
            .Cells(r, 8).Value = CDblSafe(Fld(rs, "AvgCost"))
            .Cells(r, 9).Value = NullToDate(Fld(rs, "EntryDate"))
            .Cells(r, 10).Value = NullToStr(Fld(rs, "Portfolio"))
            .Cells(r, 11).Value = NullToStr(Fld(rs, "AcctgCat"))
            .Cells(r, 14).Value = fundRate
            .Cells(r, 15).Value = NullToStr(Fld(rs, "Branch"))
            .Cells(r, 38).Value = CDblSafe(Fld(rs, "BookVal_EUR"))  ' AL
            .Cells(r, 69).Value = 1                                 ' BQ PositionSide
        End With

        bondCount = bondCount + 1
        r = r + 1
        rs.MoveNext
    Loop

    rs.Close
    conn.Close

    wsCfg.Range(CFG_LAST_LOAD).Value = Now()
    wsCfg.Range(CFG_BOND_CNT).Value = bondCount

    MsgBox CStr(bondCount) & " bonds loaded from OPICS for branch " & branchCode & ".", vbInformation
    Exit Sub

BondErr:
    MsgBox "Error loading OPICS bonds:" & vbCrLf & Err.Description, vbCritical

    On Error Resume Next
    If Not rs Is Nothing Then
        If rs.State <> 0 Then rs.Close
    End If
    If Not conn Is Nothing Then
        If conn.State <> 0 Then conn.Close
    End If
    On Error GoTo 0
End Sub


' =============================================================================
' BUTTON 2 - LOAD OPICS HEDGES
' =============================================================================

Public Sub LoadOPICS_Hedges()
    Dim wb As Workbook: Set wb = ThisWorkbook
    Dim wsCfg As Worksheet: Set wsCfg = wb.Worksheets(SH_CONFIG)
    Dim wsBnd As Worksheet: Set wsBnd = wb.Worksheets(SH_BONDS)
    Dim wsSw As Worksheet:  Set wsSw = wb.Worksheets(SH_SWAPS)
    Dim wsFut As Worksheet: Set wsFut = wb.Worksheets(SH_FUTURES)

    Update_Config_Dates_5pmUTC wsCfg
    SetupFuturesFinalHeaders wsFut
    SetupSwapsFinalHeaders wsSw

    If CleanText(wsBnd.Cells(BOND_DATA_ROW, 1).Value) = "" Then
        MsgBox "No bonds loaded. Run Button 1 first.", vbExclamation
        Exit Sub
    End If

    Dim isinType As String: isinType = CleanText(wsCfg.Range(CFG_ISIN_TYPE).Value)
    If isinType = "" Then MsgBox "Missing ISIN type in Config " & CFG_ISIN_TYPE & ".", vbExclamation: Exit Sub

    Dim asOf As String:   asOf = GetAsOfDateString(wsCfg)
    Dim allPorts As String: allPorts = "'PORT'"
    Dim isinList As String: isinList = BuildLoadedBondISINList(wsBnd)

    Dim conn As Object: Set conn = CreateObject("ADODB.Connection")
    If Not OpenOPICS(conn, wsCfg) Then Exit Sub
    Dim rs As Object: Set rs = CreateObject("ADODB.Recordset")

    On Error GoTo HedgeErr

    ' --- Swaps -------------------------------------------------------------
    rs.Open GetSwapSelectQuery(allPorts, isinList, isinType, asOf), conn, 0, 1
    wsSw.Range("A" & DATA_ROW & ":AF" & CStr(DATA_ROW + N_SWAP_ROWS - 1)).ClearContents

    Dim swapCount As Long: swapCount = 0
    Dim sr As Long: sr = DATA_ROW
    Do While Not rs.EOF And sr < DATA_ROW + N_SWAP_ROWS
        With wsSw
            .Cells(sr, 1).Value = NullToStr(Fld(rs, "DealID"))
            .Cells(sr, 2).Value = NullToStr(Fld(rs, "Ccy"))
            .Cells(sr, 3).Value = NullToNumBlank(Fld(rs, "Notional"))
            .Cells(sr, 4).Value = NullToNumBlank(Fld(rs, "FixedRate"))
            .Cells(sr, 5).Value = NullToStr(Fld(rs, "FloatIndex"))
            .Cells(sr, 6).Value = NullToNumBlank(Fld(rs, "FloatSpread"))
            .Cells(sr, 7).Value = NullToDate(Fld(rs, "StartDate"))
            .Cells(sr, 8).Value = NullToDate(Fld(rs, "EndDate"))
            .Cells(sr, 9).Value = NullToStr(Fld(rs, "PayFixed"))
            .Cells(sr, 10).Value = NullToStr(Fld(rs, "Portfolio"))
            .Cells(sr, 11).Value = NullToStr(Fld(rs, "LinkedISIN"))
            .Cells(sr, 12).Value = NullToStr(Fld(rs, "CptyName"))
            .Cells(sr, 25).Value = "Missing economics"
            .Cells(sr, 26).Value = "NONE"
        End With
        LinkSwapToBond wsBnd, CleanText(Fld(rs, "LinkedISIN")), NullToStr(Fld(rs, "DealID"))
        swapCount = swapCount + 1
        sr = sr + 1
        rs.MoveNext
    Loop
    rs.Close

    ' --- Futures -----------------------------------------------------------
    rs.Open GetFuturesSelectQuery(allPorts, isinType, asOf), conn, 0, 1
    wsFut.Range("A" & DATA_ROW & ":AI" & CStr(DATA_ROW + N_FUT_ROWS - 1)).ClearContents

    Dim futCount As Long: futCount = 0
    Dim fr As Long: fr = DATA_ROW
    Do While Not rs.EOF And fr < DATA_ROW + N_FUT_ROWS
        With wsFut
            .Cells(fr, 1).Value = NullToStr(Fld(rs, "ContractCode"))
            .Cells(fr, 2).Value = NullToStr(Fld(rs, "Exchange"))
            .Cells(fr, 3).Value = NullToStr(Fld(rs, "Currency"))
            .Cells(fr, 4).Value = CDblSafe(Fld(rs, "Contracts"))
            .Cells(fr, 5).Value = CDblSafe(Fld(rs, "FaceValue"))
            .Cells(fr, 6).Value = NullToDate(Fld(rs, "DelivDate"))
            .Cells(fr, 7).Value = NullToStr(Fld(rs, "CTD_ISIN"))
            .Cells(fr, 8).Value = CDblSafe(Fld(rs, "CTD_CF"))
            .Cells(fr, 9).Value = NullToStr(Fld(rs, "Portfolio"))
            .Cells(fr, 10).Value = NullToStr(Fld(rs, "LinkedISIN"))
            .Cells(fr, 11).Value = CDblSafe(Fld(rs, "AvgEntryPx"))
        End With
        LinkFutureToBond wsBnd, CleanText(Fld(rs, "LinkedISIN")), NullToStr(Fld(rs, "ContractCode"))
        futCount = futCount + 1
        fr = fr + 1
        rs.MoveNext
    Loop
    rs.Close
    conn.Close

    wsCfg.Range(CFG_SWAP_CNT).Value = swapCount
    wsCfg.Range(CFG_FUT_CNT).Value = futCount
    MsgBox CStr(swapCount) & " swap headers and " & CStr(futCount) & " futures loaded." & vbCrLf & _
           "Swap economics are not fabricated.", vbInformation
    Exit Sub

HedgeErr:
    MsgBox "Error loading OPICS hedges:" & vbCrLf & Err.Description, vbCritical
    On Error Resume Next
    If Not rs Is Nothing Then rs.Close
    If Not conn Is Nothing Then conn.Close
    On Error GoTo 0
End Sub

    On Error GoTo CleanFail

    Refresh_BBG_Bonds_T1_Step3C
    Refresh_BBG_Futures_T1_Step3D

    Application.EnableEvents = True
    Application.ScreenUpdating = True

    Exit Sub

    Application.Calculation = oldCalc
    Application.EnableEvents = True
    Application.ScreenUpdating = True
    MsgBox "Step 3 failed. " & eInfo, vbCritical
End Sub

    Dim ws As Worksheet:    Set ws = ThisWorkbook.Worksheets(SH_BONDS)

    If lastRow < DATA_ROW Then Exit Sub

    Dim oldCalc As XlCalculation: oldCalc = Application.Calculation
    On Error GoTo CleanFail
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual

    WriteBondSecurityCandidates ws, lastRow

    ws.Range("AR" & DATA_ROW & ":AR" & lastRow).FormulaR1C1 = BBGParseFormulaR1C1()

    ws.Range("AS" & DATA_ROW & ":AS" & lastRow).FormulaR1C1 = _
        "=IF(RC1="""","""",IF(OR(RC44="""",RC44=""UNKNOWN""),""UNKNOWN"",""OK""))"

    RefreshBloombergAndWaitMany wsCfg, ws.Range("AR" & DATA_ROW & ":AR" & lastRow)
    ws.Calculate
    LogDiagnosticsForRanges "Step3B_Resolve_Bond_Tickers", ws.Range("AR" & DATA_ROW & ":AS" & lastRow)

    Application.Calculation = oldCalc
    Application.EnableEvents = True
    Application.ScreenUpdating = True

Private Function XlBlank() As String
    ' Returns the Excel formula token "" without quote-escaping mistakes.
    XlBlank = Chr$(34) & Chr$(34)
End Function

Private Function XlText(ByVal s As String) As String
    ' Wrap text as an Excel formula string literal.
    XlText = Chr$(34) & Replace(s, Chr$(34), Chr$(34) & Chr$(34)) & Chr$(34)
End Function


' =============================================================================
' BLOOMBERG SCALABLE SECURITY-CANDIDATE FORMULA HELPERS
'
' Logic:
'   1) Candidate security strings are written in Bonds!BX:CG.
'   2) AR resolves the first parseable Bloomberg security.
'   3) Every BDP field tries AR first, then BX:CG.
' =============================================================================

Private Sub WriteBondSecurityCandidates(ByVal ws As Worksheet, ByVal lastRow As Long)
    If lastRow < DATA_ROW Then Exit Sub

    Dim f As Long
    f = DATA_ROW

    ws.Range("BX" & f & ":BX" & lastRow).FormulaR1C1 = _
        "=IF(RC1="""","""",RC1&"" ISIN"")"

    ws.Range("BY" & f & ":BY" & lastRow).FormulaR1C1 = _
        "=IF(RC1="""","""",""/isin/""&RC1)"

    ws.Range("BZ" & f & ":BZ" & lastRow).FormulaR1C1 = _
        "=IF(RC1="""","""",RC1&"" Corp"")"

    ws.Range("CA" & f & ":CA" & lastRow).FormulaR1C1 = _
        "=IF(RC1="""","""",RC1&""@BVAL Corp"")"

    ws.Range("CB" & f & ":CB" & lastRow).FormulaR1C1 = _
        "=IF(RC1="""","""",RC1&""@BGN Corp"")"

    ws.Range("CC" & f & ":CC" & lastRow).FormulaR1C1 = _
        "=IF(RC1="""","""",RC1&"" Govt"")"

    ws.Range("CD" & f & ":CD" & lastRow).FormulaR1C1 = _
        "=IF(RC1="""","""",RC1&""@BVAL Govt"")"

    ws.Range("CE" & f & ":CE" & lastRow).FormulaR1C1 = _
        "=IF(RC1="""","""",RC1&""@BGN Govt"")"

    ws.Range("CF" & f & ":CF" & lastRow).FormulaR1C1 = _
        "=IF(RC1="""","""",RC1&"" Mtge"")"

    ws.Range("CG" & f & ":CG" & lastRow).FormulaR1C1 = _
        "=IF(RC1="""","""",RC1&"" M-Mkt"")"
End Sub

Private Function BBGParseFormulaR1C1() As String
    Dim expr As String
    Dim i As Long

    expr = XlText("UNKNOWN")

    For i = COL_BBG_CAND_LAST To COL_BBG_CAND_FIRST Step -1
        expr = "IFERROR(BDP(RC" & i & ",""PARSEKYABLE_DES"")," & expr & ")"
    Next i

    BBGParseFormulaR1C1 = "=IF(RC1="""",""""," & expr & ")"
End Function

Private Function BBGTryBDPFieldExprR1C1(ByVal fieldName As String, ByVal fallbackExpr As String) As String
    Dim expr As String
    Dim i As Long

    expr = fallbackExpr

    For i = COL_BBG_CAND_LAST To COL_BBG_CAND_FIRST Step -1
        expr = "IFERROR(BDP(RC" & i & "," & XlText(fieldName) & ")," & expr & ")"
    Next i

    expr = "IFERROR(BDP(RC" & COL_BBG_TICKER & "," & XlText(fieldName) & ")," & expr & ")"

    BBGTryBDPFieldExprR1C1 = expr
End Function

' Compact variant used by MULTI-FIELD formulas.  Tries only the resolved ticker (AR=44)
' plus the two strongest candidates (/isin/ BY=77, @BVAL Corp CA=79) instead of all 11
' candidate columns.  This keeps deep multi-field formulas under Excel's hard 64-level
' function-nesting cap: full per-field x per-candidate expansion (e.g. AY's 6 fields x 11
' = 66 nested IFERROR) overflows 64 and makes .FormulaR1C1 raise run-time error 1004.
Private Function BBGTryBDPFieldExprR1C1_Short(ByVal fieldName As String, ByVal fallbackExpr As String) As String
    Dim shortCols As Variant
    Dim expr As String
    Dim k As Long

    shortCols = Array(COL_BBG_TICKER, 77, 79)   ' AR (resolved), BY (/isin/), CA (@BVAL Corp)

    expr = fallbackExpr
    For k = UBound(shortCols) To LBound(shortCols) Step -1
        expr = "IFERROR(BDP(RC" & shortCols(k) & "," & XlText(fieldName) & ")," & expr & ")"
    Next k

    BBGTryBDPFieldExprR1C1_Short = expr
End Function

Private Function BBGFirstBDPFormulaR1C1(ByVal fieldName As String, Optional ByVal fallbackExpr As String = "") As String
    If fallbackExpr = "" Then fallbackExpr = XlBlank()

    BBGFirstBDPFormulaR1C1 = _
        "=IF(RC1="""",""""," & BBGTryBDPFieldExprR1C1(fieldName, fallbackExpr) & ")"
End Function

Private Function BBGFirstBDPMultiFieldFormulaR1C1(ByVal fields As Variant, Optional ByVal fallbackExpr As String = "") As String
    Dim expr As String
    Dim i As Long

    If fallbackExpr = "" Then fallbackExpr = XlBlank()
    expr = fallbackExpr

    ' Use the compact (3-security) per-field expansion so nesting stays well under
    ' Excel's 64-level limit even for 6-field columns like AY.
    For i = UBound(fields) To LBound(fields) Step -1
        expr = BBGTryBDPFieldExprR1C1_Short(CStr(fields(i)), expr)
    Next i

    BBGFirstBDPMultiFieldFormulaR1C1 = _
        "=IF(RC1="""",""""," & expr & ")"
End Function

Private Function BBGAccruedInterestFormulaR1C1(ByVal cfgT1 As String) As String
    Dim expr As String
    Dim settleExpr As String
    Dim i As Long

    settleExpr = "TEXT(" & cfgT1 & ",""yyyymmdd"")"
    expr = "0"

    For i = COL_BBG_CAND_LAST To COL_BBG_CAND_FIRST Step -1
        expr = "IFERROR(BDP(RC" & i & ",""ACCRUED_INTEREST"",""SETTLE_DT""," & settleExpr & ")," & expr & ")"
    Next i

    expr = "IFERROR(BDP(RC" & COL_BBG_TICKER & ",""ACCRUED_INTEREST"",""SETTLE_DT""," & settleExpr & ")," & expr & ")"

    BBGAccruedInterestFormulaR1C1 = "=IF(RC1="""",""""," & expr & ")"
End Function

' Tracked formula write: records the target range in mLastFormulaTarget before the

' =============================================================================
' EFFICIENT BONDS DERIVED WRITER
'   Column map: Gov_T1=BE57 Gov_T0=BF58 Swap_T1=BG59 Swap_T0=BH60
'               OIS_T1=P16 OIS_T0=Q17  YTM_T1=Y25 YTM_T0=AF32
'               ISpread_T1=AM39 ISpread_T0=AN40
' =============================================================================

Private Sub WriteBondsCalculatedFormulas_Efficient(ws As Worksheet, lastRow As Long)
    If lastRow < DATA_ROW Then Exit Sub
    Dim f As Long: f = DATA_ROW
    Dim cfgYearsBasis As String: cfgYearsBasis = CfgR1C1(CFG_YEARSLEFT_BASIS)
    Dim cfgT1 As String:         cfgT1 = CfgR1C1(CFG_T1_DATE)

    ' O = YearsLeft
    ws.Range("O" & f & ":O" & lastRow).FormulaR1C1 = _
        "=IF(RC1="""","""",IF(" & cfgYearsBasis & "=""TODAY"",MAX(0,YEARFRAC(TODAY(),RC6,1)),MAX(0,YEARFRAC(" & cfgT1 & ",RC6,1))))"

    ' P/Q = OIS T1/T0
    ' P = OIS_T1
    ' Q = OIS_T0
    ws.Range("P" & f & ":P" & lastRow).FormulaR1C1 = _
        "=IF(RC1="""","""",IFERROR(InterpOIS(RC3,RC15,TRUE),""""))"

    ws.Range("Q" & f & ":Q" & lastRow).FormulaR1C1 = _
        "=IF(RC1="""","""",IFERROR(InterpOIS(RC3,RC15,FALSE),""""))"

    ' R/S = discount factors
    ws.Range("R" & f & ":R" & lastRow).FormulaR1C1 = "=IFERROR(EXP(-RC16*RC15),"""")"
    ws.Range("S" & f & ":S" & lastRow).FormulaR1C1 = "=IFERROR(EXP(-RC17*RC15),"""")"

    ' AV = DeltaOAS (OAS_T1=46, OAS_T0=47)
    ws.Range("AV" & f & ":AV" & lastRow).FormulaR1C1 = "=IF(AND(ISNUMBER(RC46),ISNUMBER(RC47)),RC46-RC47,"""")"

    ' BE/BF = Gov T1/T0 ; BG/BH = Swap T1/T0
    ws.Range("BE" & f & ":BE" & lastRow).FormulaR1C1 = "=IF(RC1="""","""",InterpGov(RC3,RC15,TRUE))"
    ws.Range("BF" & f & ":BF" & lastRow).FormulaR1C1 = "=IF(RC1="""","""",InterpGov(RC3,RC15,FALSE))"
    ws.Range("BG" & f & ":BG" & lastRow).FormulaR1C1 = "=IF(RC1="""","""",InterpSwap(RC3,RC15,TRUE))"
    ws.Range("BH" & f & ":BH" & lastRow).FormulaR1C1 = "=IF(RC1="""","""",InterpSwap(RC3,RC15,FALSE))"

    ' BI/BJ/BK = g = Gov - OIS  (Gov_T1=57 OIS_T1=16 ; Gov_T0=58 OIS_T0=17)
    ws.Range("BI" & f & ":BI" & lastRow).FormulaR1C1 = "=IF(AND(ISNUMBER(RC57),ISNUMBER(RC16)),RC57-RC16,"""")"
    ws.Range("BJ" & f & ":BJ" & lastRow).FormulaR1C1 = "=IF(AND(ISNUMBER(RC58),ISNUMBER(RC17)),RC58-RC17,"""")"
    ws.Range("BK" & f & ":BK" & lastRow).FormulaR1C1 = "=IF(AND(ISNUMBER(RC61),ISNUMBER(RC62)),(RC61-RC62)*10000,"""")"

' =============================================================================
' EFFICIENT FUTURES WRITERS
'   - Generic ticker via FutMap lookup (col V=22), not blind code&"1 Comdty".
'   - CTD resolved from OPICS CTD_ISIN (col G=7) into AA=27, used for CTD px/DV01.
'   - All config refs use CfgR1C1 (fixes DaysToDeliv #NAME? and futures px pull).
'   Column map: ContCode=A1 CTD_ISIN=G7 CTD_CF=H8 DelivDate=F6 Contracts=D4
'               FaceValue=E5 FX=M13 FutPx_T1=N14 FutPx_T0=Q17 CTDtkr=AA27
' =============================================================================

Private Sub WriteFutures_T1_BDP_Efficient(ws As Worksheet, lastRow As Long)
    If lastRow < DATA_ROW Then Exit Sub
    Dim f As Long: f = DATA_ROW
    Dim cfgFxFix As String:   cfgFxFix = CfgR1C1(CFG_FX_FIX_SOURCE)
    Dim cfgFutPx As String:   cfgFutPx = CfgR1C1(CFG_FUT_PRICE_FIELD)

    ' V = generic future ticker from FutMap (returns "" -> downstream flagged)
    ws.Range("V" & f & ":V" & lastRow).FormulaR1C1 = _
        "=IF(RC1="""","""",IFERROR(VLOOKUP(RC1,FutMapTable,2,FALSE),""""))"

    ' M = FX_T1  (CfgR1C1, not literal sheet concat)
    ws.Range("M" & f & ":M" & lastRow).FormulaR1C1 = _
        "=IF(RC1="""","""",IF(RC3=""EUR"",1,IFERROR(1/BDP(""EUR""&RC3&"" ""&" & cfgFxFix & "&"" Curncy"",""PX_LAST""),"""")))"

    ' N = FutPx_T1 (config field, PX_LAST fallback)
    ws.Range("N" & f & ":N" & lastRow).FormulaR1C1 = _
        "=IF(RC22="""","""",IFERROR(BDP(RC22," & cfgFutPx & "),IFERROR(BDP(RC22,""PX_LAST""),"""")))"

    ws.Range("Y" & f & ":Y" & lastRow).FormulaR1C1 = "=IF(RC22="""","""",IFERROR(BDP(RC22,""FUT_VAL_PT""),""""))"
    ws.Range("Z" & f & ":Z" & lastRow).FormulaR1C1 = "=IF(RC22="""","""",IFERROR(BDP(RC22,""FUT_CONT_SIZE""),""""))"

    ' AA = CTD ticker resolved from OPICS CTD_ISIN (col 7), NOT Bloomberg FUT_CTD_BOND
    ws.Range("AA" & f & ":AA" & lastRow).FormulaR1C1 = _
        "=IF(RC7="""",""""," & _
        "IFERROR(BDP(""/isin/""&RC7,""PARSEKYABLE_DES"")," & _
        "IFERROR(BDP(RC7&"" Govt"",""PARSEKYABLE_DES"")," & _
        "IFERROR(BDP(RC7&"" Corp"",""PARSEKYABLE_DES""),""""))))"

    ' AB = CTD unit DV01 from CTD ticker (AA=27)
    ws.Range("AB" & f & ":AB" & lastRow).FormulaR1C1 = _
        "=IF(RC27="""",""""," & _
        "IFERROR(BDP(RC27,""DV01"")," & _
        "IF(AND(ISNUMBER(BDP(RC27,""DUR_ADJ_MID"")),ISNUMBER(RC15)),BDP(RC27,""DUR_ADJ_MID"")*RC15*0.0001,"""")))"

    ' AC = conversion factor: use OPICS CTD_CF (col 8); Bloomberg CONV_FACTOR fallback
    ws.Range("AC" & f & ":AC" & lastRow).FormulaR1C1 = _
        "=IF(ISNUMBER(RC8),RC8,IF(RC22="""","""",IFERROR(BDP(RC22,""CONV_FACTOR""),"""")))"

    ' O = CTD dirty price T1 (from CTD ticker AA=27)
    ws.Range("O" & f & ":O" & lastRow).FormulaR1C1 = "=IF(RC27="""","""",IFERROR(BDP(RC27,""PX_DIRTY_MID""),""""))"
    ' P = CTD ticker mirror for T0 BDH targeting
    ws.Range("P" & f & ":P" & lastRow).FormulaR1C1 = "=IF(RC27="""","""",RC27)"

    ' Basis diagnostics (best-effort)
    ws.Range("AF" & f & ":AF" & lastRow).FormulaR1C1 = "=IF(RC22="""","""",IFERROR(BDP(RC22,""IMPLIED_REPO_RATE""),""""))"
    ws.Range("AG" & f & ":AG" & lastRow).FormulaR1C1 = "=IF(RC22="""","""",IFERROR(BDP(RC22,""NET_BASIS""),""""))"
    ws.Range("AH" & f & ":AH" & lastRow).FormulaR1C1 = "=IF(RC22="""","""",IFERROR(BDP(RC22,""GROSS_BASIS""),""""))"

    ' Status
    ws.Range("W" & f & ":W" & lastRow).FormulaR1C1 = "=IF(RC14="""",""Missing T1 future price"",""OK"")"
    ws.Range("AI" & f & ":AI" & lastRow).FormulaR1C1 = _
        "=IF(RC1="""","""",IF(RC22="""",""Unmapped contract (FutMap)"",IF(RC14="""",""Missing T1 future price"",""OK"")))"
End Sub

' =============================================================================
' T0 BDH STAGING SHEET
'
'   Why:  BDH() is an array (Date|Value) function.  Writing ~4,800 of them straight
'   into the live Bonds/Futures rows (interleaved with calc columns such as the AI
'   DV01 that sits between AH and AJ) is fragile and re-hits Bloomberg on every
'   recalc.  Instead we stage all single-point T0 pulls on ONE hidden support sheet,
'   refresh once, then copy only the resolved values back into the model T0 columns.
'
'   The sheet is created once and REUSED on every run (looked up by name first; only
'   added if missing), so re-running Step 4 never spawns "BBG_T0_Staging (2)" etc.
'
'   Layout: staging rows align 1:1 to model rows (row 5..lastRow).  Each staging cell
'   reuses BDHLastPointExprR1C1 (single value, non-spilling) with the security cell
'   sheet-qualified back to the model via cross-sheet R1C1 ('Bonds'!RC44 etc.).
'
'   Staging col -> model target:
'     B -> Bonds U  (FX_T0)            C -> Bonds AD (CleanPx_T0, PX_LAST)
'     D -> Bonds AE (DirtyPx_T0)       E -> Bonds AF (YTM_T0)
'     F -> Bonds AG (ZSprd_T0)         G -> Bonds AH (ASW_T0)
'     H -> Bonds AU (OAS_T0)           I -> Bonds BT (FundRate_T0)
'     K -> Futures Q (FutPx_T0)
' =============================================================================

' Create-once / reuse the hidden staging sheet.  Returned visible (so the Bloomberg
' add-in calculates it); caller re-hides it (xlSheetVeryHidden) when done.
Private Function EnsureT0StagingSheet() As Worksheet
    Dim ws As Worksheet
    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets(SH_T0_STG)
    On Error GoTo 0

    If ws Is Nothing Then
        Set ws = ThisWorkbook.Worksheets.Add(After:=ThisWorkbook.Worksheets(ThisWorkbook.Worksheets.Count))
        ws.Name = SH_T0_STG
    End If

    ws.Visible = xlSheetVisible
    ws.Cells.ClearContents
    ws.Range("A1").Value = "BBG_T0_Staging (auto-generated; do not edit)"

    Set EnsureT0StagingSheet = ws
End Function

' Builds a single-point staging formula:  =IF(secRef="","",<last BDH value>)
'   secRef    : sheet-qualified R1C1 security cell, e.g. "'Bonds'!RC44"
'   fieldExpr : Excel expression for the field (a quoted literal or a config cell ref)
Private Function BDHStgPointR1C1(ByVal secRef As String, ByVal fieldExpr As String, ByVal cfgT0 As String) As String
    BDHStgPointR1C1 = _
        "=IF(" & secRef & "="""",""""," & _
        BDHLastPointExprR1C1( _
            secRef, _
            fieldExpr, _
            "INT(" & cfgT0 & ")-3", _
            "INT(" & cfgT0 & ")" _
        ) & ")"
End Function

' Writes the bond T0 BDH grid onto the staging sheet; returns the populated range.
Private Function WriteBondT0Staging(ByVal wsStg As Worksheet, ByVal lastBondRow As Long) As Range
    If lastBondRow < DATA_ROW Then Exit Function

    Dim cfgFxFix As String: cfgFxFix = CfgR1C1(CFG_FX_FIX_SOURCE)
    Dim cfgT0 As String:    cfgT0 = CfgR1C1(CFG_T0_DATE)
    Dim f As Long:          f = DATA_ROW

    ' B = FX_T0  (1 for EUR; else 1/BDH(EUR{ccy}{fix} Curncy, PX_LAST); blank on miss)
    wsStg.Range("B" & f & ":B" & lastBondRow).FormulaR1C1 = _
        "=IF('" & SH_BONDS & "'!RC1="""","""",IF('" & SH_BONDS & "'!RC3=""EUR"",1,IFERROR(1/" & _
        BDHLastPointExprR1C1( _
            """EUR""&'" & SH_BONDS & "'!RC3&"" ""&" & cfgFxFix & "&"" Curncy""", _
            """PX_LAST""", _
            "INT(" & cfgT0 & ")-3", _
            "INT(" & cfgT0 & ")" _
        ) & ","""")))"

    Dim secAR As String: secAR = "'" & SH_BONDS & "'!RC44"   ' resolved BBG ticker
    Dim secBR As String: secBR = "'" & SH_BONDS & "'!RC70"   ' funding ticker

    wsStg.Range("C" & f & ":C" & lastBondRow).FormulaR1C1 = BDHStgPointR1C1(secAR, """" & BBG_PX_LAST & """", cfgT0)
    wsStg.Range("D" & f & ":D" & lastBondRow).FormulaR1C1 = BDHStgPointR1C1(secAR, """" & BBG_PX_DIRTY_MID & """", cfgT0)
    wsStg.Range("E" & f & ":E" & lastBondRow).FormulaR1C1 = BDHStgPointR1C1(secAR, """" & BBG_YLD_YTM_MID & """", cfgT0)
    wsStg.Range("F" & f & ":F" & lastBondRow).FormulaR1C1 = BDHStgPointR1C1(secAR, """" & BBG_ZSPREAD & """", cfgT0)
    wsStg.Range("G" & f & ":G" & lastBondRow).FormulaR1C1 = BDHStgPointR1C1(secAR, """" & BBG_ASW & """", cfgT0)
    wsStg.Range("H" & f & ":H" & lastBondRow).FormulaR1C1 = BDHStgPointR1C1(secAR, """" & BBG_OAS & """", cfgT0)
    wsStg.Range("I" & f & ":I" & lastBondRow).FormulaR1C1 = BDHStgPointR1C1(secBR, """" & BBG_PX_LAST & """", cfgT0)

    Set WriteBondT0Staging = wsStg.Range("B" & f & ":I" & lastBondRow)
End Function

' Writes the futures T0 BDH column onto the staging sheet; returns the populated range.
Private Function WriteFutureT0Staging(ByVal wsStg As Worksheet, ByVal lastFutRow As Long) As Range
    If lastFutRow < DATA_ROW Then Exit Function

    Dim cfgFutPx As String: cfgFutPx = CfgR1C1(CFG_FUT_PRICE_FIELD)
    Dim cfgT0 As String:    cfgT0 = CfgR1C1(CFG_T0_DATE)
    Dim f As Long:          f = DATA_ROW

    ' K = FutPx_T0  (generic future ticker in Futures!V=col22; config price field)
    wsStg.Range("K" & f & ":K" & lastFutRow).FormulaR1C1 = _
        BDHStgPointR1C1("'" & SH_FUTURES & "'!RC22", cfgFutPx, cfgT0)

    Set WriteFutureT0Staging = wsStg.Range("K" & f & ":K" & lastFutRow)
End Function

' Copies the resolved staging values back into the model T0 columns (value blocks,
' so blanks stay "" and the missing-as-blank contract is preserved).
Private Sub CopyT0StagingToModel(ByVal wsStg As Worksheet, ByVal wsBnd As Worksheet, _
                                 ByVal wsFut As Worksheet, ByVal lastBondRow As Long, _
                                 ByVal lastFutRow As Long)
    If lastBondRow >= DATA_ROW Then
        wsBnd.Range("U" & DATA_ROW & ":U" & lastBondRow).Value = wsStg.Range("B" & DATA_ROW & ":B" & lastBondRow).Value
        wsBnd.Range("AD" & DATA_ROW & ":AD" & lastBondRow).Value = wsStg.Range("C" & DATA_ROW & ":C" & lastBondRow).Value
        wsBnd.Range("AE" & DATA_ROW & ":AE" & lastBondRow).Value = wsStg.Range("D" & DATA_ROW & ":D" & lastBondRow).Value
        wsBnd.Range("AF" & DATA_ROW & ":AF" & lastBondRow).Value = wsStg.Range("E" & DATA_ROW & ":E" & lastBondRow).Value
        wsBnd.Range("AG" & DATA_ROW & ":AG" & lastBondRow).Value = wsStg.Range("F" & DATA_ROW & ":F" & lastBondRow).Value
        wsBnd.Range("AH" & DATA_ROW & ":AH" & lastBondRow).Value = wsStg.Range("G" & DATA_ROW & ":G" & lastBondRow).Value
        wsBnd.Range("AU" & DATA_ROW & ":AU" & lastBondRow).Value = wsStg.Range("H" & DATA_ROW & ":H" & lastBondRow).Value
        wsBnd.Range("BT" & DATA_ROW & ":BT" & lastBondRow).Value = wsStg.Range("I" & DATA_ROW & ":I" & lastBondRow).Value
    End If

    If lastFutRow >= DATA_ROW Then
        wsFut.Range("Q" & DATA_ROW & ":Q" & lastFutRow).Value = wsStg.Range("K" & DATA_ROW & ":K" & lastFutRow).Value
    End If
End Sub

' =============================================================================
' PNL ATTRIBUTION
'   Full header + row rebuild removes any stale #REF! formulas in the sheet.
'   Run RebuildPNLOnly for a fast repair without touching Bloomberg.
' =============================================================================

Public Sub RebuildPNLOnly()
    Dim wsPnl As Worksheet: Set wsPnl = ThisWorkbook.Worksheets(SH_PNL)
    Dim wsBnd As Worksheet: Set wsBnd = ThisWorkbook.Worksheets(SH_BONDS)
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    SetupPNLAttributionHeaders wsPnl
    WritePNLAttributionRows wsPnl, wsBnd
    Application.CalculateFull
    Application.EnableEvents = True
    Application.ScreenUpdating = True
    MsgBox "PNL_Attribution rebuilt (stale #REF! cleared).", vbInformation
End Sub


Private Sub SetupPNLAttributionHeaders(ws As Worksheet)
    ' Clear the whole data + header block first to kill any #REF! remnants.
    On Error Resume Next
    ws.Range("A3:CA3").UnMerge
    On Error GoTo 0
    ws.Range("A3:CA" & CStr(DATA_ROW + N_BOND_ROWS - 1)).ClearContents

    ws.Range("A1").Value = "PNL Attribution"

End Sub

Private Function BDHLastPointExprR1C1( _
    ByVal secExpr As String, _
    ByVal fieldExpr As String, _
    ByVal startDateExpr As String, _
    ByVal endDateExpr As String) As String

    ' -------------------------------------------------------------------------
    ' Safer BDH single-point extractor.
    '
    ' OLD VERSION USED:
    '   "Days=W","Fill=P","UseDPDF=N","Dts=H","Cols=N"
    '
    ' That is dangerous because UseDPDF / Dts / Cols can be interpreted by
    ' Bloomberg as invalid override field IDs depending on the Excel API version.
    '
    ' NEW VERSION:
    '   Keeps only safer BDH options:
    '       "Days=W"
    '       "Fill=P"
    '       "Sort=A"
    '
    ' Then it handles either:
    '   - normal BDH 2-column output: Date | Value
    '   - compressed/single-column output, if Bloomberg returns that
    '
    ' It returns the last available value in the BDH result.
    ' -------------------------------------------------------------------------

    BDHLastPointExprR1C1 = _
        "IFERROR(" & _
            "LET(" & _
                "_bdh,BDH(" & secExpr & "," & fieldExpr & "," & _
                    startDateExpr & "," & endDateExpr & "," & _
                    """Days=W"",""Fill=P"",""Sort=A"")," & _
                "_r,ROWS(_bdh)," & _
                "_c,COLUMNS(_bdh)," & _
                "IF(_c>=2,INDEX(_bdh,_r,2),INDEX(_bdh,_r,1))" & _
            ")," & _
        """"")"

End Function

' Progress-aware readiness wait.
'
'   The old version used a FLAT deadline: if any cell was still "#N/A Requesting Data"
'   when CFG_BBG_TIMEOUT elapsed it failed - even though the data was still streaming in
'   and finished moments later.  That false-aborted Step 4 on large BDH batches (117
'   curve cells, and 4,800 bond/future staging cells).
'
'   This version waits as long as Bloomberg keeps making PROGRESS (the count of pending
'   cells keeps dropping) and only fails on:
'     * a hard session error (Connection / Authorization / Invalid override), or
'     * a true STALL (no progress for CFG_BBG_TIMEOUT seconds), or
'     * an absolute cap (safety, for genuinely stuck batches).
'   A final settle + authoritative re-scan closes the race where the last cells resolve
'   just as the loop ends.
Private Function WaitForBloombergReadyMany(ByVal wsCfg As Worksheet, ParamArray rngs() As Variant) As Boolean
    Dim stallSec As Long
    stallSec = val(CleanText(wsCfg.Range(CFG_BBG_TIMEOUT).Value))
    If stallSec <= 0 Then stallSec = 120
    If stallSec < 30 Then stallSec = 30

    Dim capSec As Long
    capSec = stallSec * 6
    If capSec < 600 Then capSec = 600          ' absolute safety cap (>= 10 min)

    Dim tStart As Single:    tStart = Timer
    Dim tProgress As Single: tProgress = Timer
    Dim prevPending As Long: prevPending = -1
    Dim curPending As Long

    Dim rngArr As Variant
    rngArr = rngs

    Do
        DoEvents
        Application.Calculate

        ' Genuine session failure -> stop now (data is not actually coming).
        If RangeHasHardBBGError(rngArr) Then
            WaitForBloombergReadyMany = False
            Exit Function
        End If

        curPending = CountPendingBBG(rngArr)
        If curPending = 0 Then Exit Do                       ' everything settled

        ' Progress resets the stall timer; large batches keep getting time as long as
        ' cells keep clearing.
        If prevPending < 0 Or curPending < prevPending Then tProgress = Timer
        prevPending = curPending

        If (Timer - tProgress) > stallSec Then Exit Do       ' stalled (no progress)
        If (Timer - tStart) > capSec Then Exit Do            ' absolute cap

        WaitSeconds 1
    Loop

    ' Final settle, then make the authoritative decision from actual cell state.
    DoEvents
    Application.Calculate
    WaitSeconds 1
    DoEvents
    Application.Calculate

    If CountPendingBBG(rngArr) > 0 Then
        WaitForBloombergReadyMany = False
        Exit Function
    End If

    If RangeHasHardBBGError(rngArr) Then
        WaitForBloombergReadyMany = False
        Exit Function
    End If

    WaitForBloombergReadyMany = True
End Function

' Total number of cells still showing a Bloomberg "Requesting"/pending state.
Private Function CountPendingBBG(ByVal rngs As Variant) As Long
    Dim i As Long, n As Long
    Dim rg As Range
    n = 0
    For i = LBound(rngs) To UBound(rngs)
        If TypeName(rngs(i)) = "Range" Then
            Set rg = rngs(i)
            If RangeHasPendingBBG(rg) Then
                Dim c As Range
                For Each c In rg.Cells
                    If InStr(1, CStr(c.Text), "Requesting", vbTextCompare) > 0 Then n = n + 1
                Next c
            End If
        End If
    Next i
    CountPendingBBG = n
End Function

' True only for HARD Bloomberg errors that will not resolve by waiting (session/auth/
' invalid override).  Transient "Requesting" is deliberately excluded.
Private Function RangeHasHardBBGError(ByVal rngs As Variant) As Boolean
    Dim i As Long
    Dim rg As Range
    Dim c As Range
    Dim txt As String
    For i = LBound(rngs) To UBound(rngs)
        If TypeName(rngs(i)) = "Range" Then
            Set rg = rngs(i)
            For Each c In rg.Cells
                txt = CStr(c.Text)
                If InStr(1, txt, "#N/A Connection", vbTextCompare) > 0 _
                   Or InStr(1, txt, "Authorization", vbTextCompare) > 0 _
                   Or InStr(1, txt, "Invalid override field id", vbTextCompare) > 0 Then
                    Debug.Print "Hard Bloomberg error at " & c.Worksheet.Name & "!" & c.Address(False, False) & " -> " & txt
                    RangeHasHardBBGError = True
                    Exit Function
                End If
            Next c
        End If
    Next i
    RangeHasHardBBGError = False
End Function

Private Function RangeHasUnresolvedBBGProblem(ByVal rng As Range) As Boolean

    Dim c As Range
    Dim txt As String

    For Each c In rng.Cells

        txt = CStr(c.Text)

        If InStr(1, txt, "Requesting", vbTextCompare) > 0 Then
            RangeHasUnresolvedBBGProblem = True
            Exit Function
        End If

        If InStr(1, txt, "#N/A Requesting", vbTextCompare) > 0 Then
            RangeHasUnresolvedBBGProblem = True
            Exit Function
        End If

        If InStr(1, txt, "#N/A Connection", vbTextCompare) > 0 Then
            RangeHasUnresolvedBBGProblem = True
            Exit Function
        End If

        If InStr(1, txt, "#N/A Invalid", vbTextCompare) > 0 Then
            Debug.Print "Bloomberg invalid formula at " & c.Worksheet.Name & "!" & c.Address(False, False)
            Debug.Print c.FormulaR1C1

            RangeHasUnresolvedBBGProblem = True
            Exit Function
        End If

        If InStr(1, txt, "Invalid override field id", vbTextCompare) > 0 Then
            Debug.Print "Invalid override field ID at " & c.Worksheet.Name & "!" & c.Address(False, False)
            Debug.Print c.FormulaR1C1

            RangeHasUnresolvedBBGProblem = True
            Exit Function
        End If

        If InStr(1, txt, "Authorization", vbTextCompare) > 0 Then
            RangeHasUnresolvedBBGProblem = True
            Exit Function
        End If

    Next c

    RangeHasUnresolvedBBGProblem = False

End Function

Private Function WriteCurveT0BDHFormulas_OptionB(ws As Worksheet) As Range
    Dim outCols As Variant
    Dim tickerCols As Variant
    Dim snapRows As Variant

    outCols = Array("A", "D", "G", "R", "U", "Y", "AI", "AL", "AO")
    tickerCols = Array("C", "F", "I", "T", "W", "Z", "AK", "AN", "AQ")

    snapRows = Array(51, 56, 62, 67, 72, 77, 82, 87, 92, 97, 102, 107, 112)

    Dim blockIdx As Long
    Dim tenorIdx As Long
    Dim tickerColNum As Long
    Dim outColNum As Long
    Dim tickerRow As Long
    Dim outRow As Long

    Dim outCell As Range
    Dim targetRng As Range

    For blockIdx = LBound(outCols) To UBound(outCols)

        tickerColNum = ws.Range(CStr(tickerCols(blockIdx)) & "1").Column
        outColNum = ws.Range(CStr(outCols(blockIdx)) & "1").Column

        For tenorIdx = LBound(snapRows) To UBound(snapRows)

            tickerRow = 7 + tenorIdx
            outRow = CLng(snapRows(tenorIdx))

            Set outCell = ws.Cells(outRow, outColNum)

            outCell.FormulaR1C1 = _
                "=IF(R" & tickerRow & "C" & tickerColNum & "="""",""""," & _
                BDHLastPointExprR1C1( _
                    "R" & tickerRow & "C" & tickerColNum, _
                    """PX_LAST""", _
                    "R46C2", _
                    "R47C2" _
                ) & ")"

            If targetRng Is Nothing Then
                Set targetRng = outCell
            Else
                Set targetRng = Application.Union(targetRng, outCell)
            End If

        Next tenorIdx
    Next blockIdx

    Set WriteCurveT0BDHFormulas_OptionB = targetRng
End Function


' =============================================================================
' BUTTON 4 - FREEZE T0 SNAPSHOT
'   Writes BDH formulas, refreshes, then converts T0 ranges to static values.
'   Missing-as-blank is preserved by the BDH writers (they return "").
' =============================================================================

Public Sub Freeze_T0_Snapshot_Step4()
    Dim wb As Workbook: Set wb = ThisWorkbook
    Dim wsCfg As Worksheet: Set wsCfg = wb.Worksheets(SH_CONFIG)
    Dim wsOIS As Worksheet: Set wsOIS = wb.Worksheets(SH_OIS)
    Dim wsBnd As Worksheet: Set wsBnd = wb.Worksheets(SH_BONDS)
    Dim wsFut As Worksheet: Set wsFut = wb.Worksheets(SH_FUTURES)

    Dim oldCalc As XlCalculation
    oldCalc = Application.Calculation

    On Error GoTo CleanFail

    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual

    Dim t0DateTime As Date
    t0DateTime = GetOrSetT0SnapshotDateTime(wsCfg)

    wsOIS.Range("A44").Value = "T0 curve snapshot datetime"
    wsOIS.Range("B44").Value = t0DateTime
    wsOIS.Range("B44").NumberFormat = "dd/mm/yyyy hh:mm"

    wsOIS.Range("A46").Value = "BDH start date"
    wsOIS.Range("B46").Value = PrevBusinessDay(DateValue(t0DateTime))
    wsOIS.Range("B46").NumberFormat = "dd/mm/yyyy"

    wsOIS.Range("A47").Value = "BDH end date"
    wsOIS.Range("B47").Value = DateValue(t0DateTime)
    wsOIS.Range("B47").NumberFormat = "dd/mm/yyyy"

    wsOIS.Range("A48").Value = "Last frozen at"
    wsOIS.Range("B48").Value = Now()
    wsOIS.Range("B48").NumberFormat = "dd/mm/yyyy hh:mm:ss"

    Dim curveStagingRng As Range
    Set curveStagingRng = WriteCurveT0BDHFormulas_OptionB(wsOIS)

    RefreshBloombergAndWaitMany wsCfg, curveStagingRng

    If Not WaitForBloombergReadyMany(wsCfg, curveStagingRng) Then
        Err.Raise 9510, , _
            "Bloomberg curve T0 requests did not finish cleanly. " & _
            "The macro stopped before freezing to avoid storing #N/A Requesting Data."
    End If

    wsOIS.Calculate

    FreezeRange curveStagingRng
    CopyCurveT0StagingToModelInputs_OptionB wsOIS

    Dim lastBondRow As Long
    lastBondRow = LastBondDataRow(wsBnd)

    If lastBondRow >= DATA_ROW Then
        WriteBondsCalculatedFormulas_Efficient wsBnd, lastBondRow
    End If

    Dim lastFutRow As Long
    lastFutRow = LastFutureDataRow(wsFut)

    ' --- T0 market pulls on the hidden staging sheet (one BDH grid, refreshed once) ---
    Dim wsStg As Worksheet
    Set wsStg = EnsureT0StagingSheet()

    Dim bondStgRng As Range, futStgRng As Range
    Set bondStgRng = WriteBondT0Staging(wsStg, lastBondRow)
    Set futStgRng = WriteFutureT0Staging(wsStg, lastFutRow)

    RefreshBloombergAndWaitMany wsCfg, bondStgRng, futStgRng

    If Not WaitForBloombergReadyMany(wsCfg, bondStgRng, futStgRng) Then
        ' Leave the staging sheet visible so the raw BDH output can be inspected.
        Err.Raise 9511, , _
            "Bloomberg bond/future T0 requests did not finish cleanly. " & _
            "Inspect sheet '" & SH_T0_STG & "' for the raw BDH outputs. " & _
            "The macro stopped before freezing to avoid storing #N/A Requesting Data."
    End If

    wsStg.Calculate

    ' Copy resolved values back into the model, then derive AN/AP and futures status.
    CopyT0StagingToModel wsStg, wsBnd, wsFut, lastBondRow, lastFutRow

    If lastBondRow >= DATA_ROW Then WriteBonds_T0_DerivedSpreads wsBnd, lastBondRow
    If lastFutRow >= DATA_ROW Then WriteFutures_T0_Status wsFut, lastFutRow

    wsOIS.Calculate
    wsBnd.Calculate
    wsFut.Calculate

    FreezeT0Ranges wsOIS, wsBnd, wsFut

    ' Re-hide the staging sheet now that values are copied and frozen.
    wsStg.Visible = xlSheetVeryHidden

    wsCfg.Range(CFG_LAST_BBG).Value = Now()

    Application.Calculation = oldCalc
    Application.EnableEvents = True
    Application.ScreenUpdating = True

    MsgBox "Step 4 complete: T0 snapshot frozen." & vbCrLf & _
           "T0 datetime used: " & Format(t0DateTime, "dd/mm/yyyy hh:mm"), _
           vbInformation

    Exit Sub

CleanFail:
    Dim eInfo As String: eInfo = CaptureErrInfo()
    Application.Calculation = oldCalc
    Application.EnableEvents = True
    Application.ScreenUpdating = True

    MsgBox "Step 4 failed:" & vbCrLf & eInfo, vbCritical
End Sub

Private Function GetBondSelectQuery( _
    ByVal allInvTypes As String, _
    ByVal isinType As String, _
    ByVal asOf As String, _
    Optional ByVal branchCode As String = "") As String
    Dim s As String
    s = s & "SELECT isin.SECALTID AS ISIN,"
    s = s & " sm.DESCR AS Name,"
    s = s & " sp.CCY AS Currency,"
    s = s & " sm.COUPRATE_8 AS Coupon_Dec,"
    s = s & " sm.INTPAYCYCLE AS CouponFreq,"
    s = s & " sm.MDATE AS MaturityDate,"
    s = s & " sp.QTY AS Notional,"
    s = s & " CASE"
    s = s & " WHEN ABS(ISNULL(sp.QTY,0))>0 AND ABS(ISNULL(tp.TDBOOKVALUE,0))>0 THEN ABS(tp.TDBOOKVALUE)/NULLIF(ABS(sp.QTY),0)*100"
    s = s & " WHEN ABS(ISNULL(sp.QTY,0))>0 AND ABS(ISNULL(tp.ADJTDBOOKVAL,0))>0 THEN ABS(tp.ADJTDBOOKVAL)/NULLIF(ABS(sp.QTY),0)*100"
    s = s & " ELSE ISNULL(sp.PURCHAVGCOST,0) END AS AvgCost,"
    s = s & " sp.SETTDATE AS EntryDate,"
    s = s & " sp.PORT AS Portfolio,"
    s = s & " LTRIM(RTRIM(sp.INVTYPE)) AS AcctgCat,"
    s = s & " sp.BR AS Branch,"
    s = s & " sm.PRODUCT AS Product,"
    s = s & " ISNULL(rp.REPORATE_8,0) AS FundingRate_Dec,"
    s = s & " ISNULL(tp.TDBOOKVALUE,0) AS BookVal_EUR"
    s = s & " FROM dbo.SPOS sp"
    s = s & " INNER JOIN dbo.SECM sm ON sm.SECID=sp.SECID"
    s = s & " LEFT JOIN dbo.ASID isin ON isin.SECID=sp.SECID AND isin.SECIDTYPE='" & EscapeSQL(isinType) & "'"
    s = s & " LEFT JOIN dbo.TPOS tp ON tp.BR=sp.BR AND tp.COST=sp.COST AND tp.PORT=sp.PORT AND tp.SECID=sp.SECID AND tp.INVTYPE=sp.INVTYPE"
    s = s & " LEFT JOIN (SELECT r2.SECID,r2.PORT,AVG(r2.REPORATE_8) AS REPORATE_8 FROM dbo.RPDT r2"
    s = s & " WHERE r2.VDATE<='" & asOf & "' AND r2.MDATE>='" & asOf & "' GROUP BY r2.SECID,r2.PORT) rp"
    s = s & " ON rp.SECID=sp.SECID AND rp.PORT=sp.PORT"
    s = s & " WHERE sp.QTY<>0"
    s = s & " AND sp.PORT='PORT'"

    If CleanText(branchCode) <> "" Then
        s = s & " AND LTRIM(RTRIM(sp.BR))='" & EscapeSQL(branchCode) & "'"
    End If

    s = s & " AND LTRIM(RTRIM(sp.INVTYPE)) IN (" & allInvTypes & ")"
    s = s & " AND LTRIM(RTRIM(sp.INVTYPE)) <> 'T'"
    s = s & " AND LTRIM(RTRIM(sm.PRODUCT)) IN ('OBR','SECUR')"
    s = s & " ORDER BY sp.CCY,LTRIM(RTRIM(sp.INVTYPE)),sp.COST,sm.MDATE"
    GetBondSelectQuery = s
End Function

Private Function GetSwapSelectQuery(allPorts As String, isinList As String, isinType As String, asOf As String) As String
    Dim portFilter As String, isinFilter As String, whereClause As String, s As String
    If allPorts <> "" Then portFilter = "LTRIM(RTRIM(sw.PORT)) IN (" & allPorts & ")"
    If isinList <> "" Then isinFilter = "sw.UNDSECID IN (SELECT SECID FROM dbo.ASID WHERE SECALTID IN (" & isinList & ") AND SECIDTYPE='" & EscapeSQL(isinType) & "')"

    If portFilter <> "" And isinFilter <> "" Then
        whereClause = "(" & portFilter & " OR " & isinFilter & ")"
    ElseIf portFilter <> "" Then
        whereClause = portFilter
    ElseIf isinFilter <> "" Then
        whereClause = isinFilter
    Else
        whereClause = "1=1"
    End If

    s = s & "SELECT sw.DEALNO AS DealID,"
    s = s & " NULL AS Ccy,NULL AS Notional,NULL AS FixedRate,NULL AS FloatIndex,NULL AS FloatSpread,"
    s = s & " sw.STARTDATE AS StartDate,sw.MATDATE AS EndDate,NULL AS PayFixed,sw.PORT AS Portfolio,"
    s = s & " (SELECT TOP 1 a2.SECALTID FROM dbo.ASID a2 WHERE a2.SECID=sw.UNDSECID AND a2.SECIDTYPE='" & EscapeSQL(isinType) & "') AS LinkedISIN,"
    s = s & " c.SN AS CptyName"
    s = s & " FROM dbo.SWDH sw"
    s = s & " LEFT JOIN dbo.CUST c ON c.CNO=sw.CNO"
    s = s & " WHERE LTRIM(RTRIM(CONVERT(varchar(10),sw.VERIND)))='1'"
    s = s & " AND LTRIM(RTRIM(sw.PRODUCT))='SWAP'"
    s = s & " AND sw.MATDATE>'" & asOf & "'"
    s = s & " AND " & whereClause
    s = s & " ORDER BY sw.PORT,sw.MATDATE"
    GetSwapSelectQuery = s
End Function

Private Function GetFuturesSelectQuery(allPorts As String, isinType As String, asOf As String) As String
    Dim s As String
    s = s & "WITH CTD AS (SELECT BR,CONTCODE,DELVDATE,CONVFACTOR_8,UNDSECID,"
    s = s & " ROW_NUMBER() OVER(PARTITION BY BR,CONTCODE ORDER BY DELVDATE) AS RN"
    s = s & " FROM dbo.FDEL WHERE DELVDATE>='" & asOf & "'),"
    s = s & " AE AS (SELECT BR,PORT,COST,CONTCODE,"
    s = s & " SUM(NUMCONT*CONTPRICE_8)/NULLIF(SUM(ABS(NUMCONT)),0) AS AvgPx"
    s = s & " FROM dbo.FFDH WHERE VERIND='Y' AND (REVREASON IS NULL OR REVREASON='')"
    s = s & " GROUP BY BR,PORT,COST,CONTCODE)"
    s = s & " SELECT fp.CONTCODE AS ContractCode,fc.EXCHANGE AS Exchange,fc.CCY AS Currency,"
    s = s & " fp.NUMCONT AS Contracts,fc.FACEVALUE AS FaceValue,ctd.DELVDATE AS DelivDate,"
    s = s & " isin.SECALTID AS CTD_ISIN,ctd.CONVFACTOR_8 AS CTD_CF,fp.PORT AS Portfolio,"
    s = s & " (SELECT TOP 1 a3.SECALTID FROM dbo.ASID a3 WHERE a3.SECID=("
    s = s & " SELECT TOP 1 fh2.UNDSECID FROM dbo.FFDH fh2 WHERE fh2.BR=fp.BR AND fh2.CONTCODE=fp.CONTCODE"
    s = s & " AND fh2.UNDSECID IS NOT NULL AND fh2.UNDSECID<>'' ORDER BY fh2.DEALDATE DESC)"
    s = s & " AND a3.SECIDTYPE='" & EscapeSQL(isinType) & "') AS LinkedISIN,"
    s = s & " ae.AvgPx AS AvgEntryPx"
    s = s & " FROM dbo.FPOS fp"
    s = s & " INNER JOIN dbo.FCON fc ON fc.BR=fp.BR AND fc.CONTCODE=fp.CONTCODE AND fc.PRODUCT=fp.PRODUCT"
    s = s & " LEFT JOIN CTD ctd ON ctd.BR=fp.BR AND ctd.CONTCODE=fp.CONTCODE AND ctd.RN=1"
    s = s & " LEFT JOIN dbo.ASID isin ON isin.SECID=ctd.UNDSECID AND isin.SECIDTYPE='" & EscapeSQL(isinType) & "'"
    s = s & " LEFT JOIN AE ae ON ae.BR=fp.BR AND ae.PORT=fp.PORT AND ae.CONTCODE=fp.CONTCODE"
    s = s & " WHERE fp.NUMCONT<>0 AND fp.PORT IN (" & allPorts & ")"
    s = s & " ORDER BY fc.CCY,fp.PORT,ctd.DELVDATE"
    GetFuturesSelectQuery = s
End Function

Public Function InterpOIS(ccy As String, years As Double, useT1 As Boolean) As Variant

Private Function InterpCurveValue(ccy As String, curveType As String, years As Double, useT1 As Boolean) As Variant
    On Error GoTo ErrHandler
    Dim ws As Worksheet: Set ws = ThisWorkbook.Worksheets(SH_OIS)
    Dim yRange As Range, rRange As Range

    Select Case UCase(Trim(ccy))
        Case "EUR"
            Set yRange = ws.Range("B7:B19")
            If curveType = "OIS" Then
                If useT1 Then Set rRange = ws.Range("E7:E19") Else Set rRange = ws.Range("D7:D19")
            ElseIf curveType = "GOV" Then
                If useT1 Then Set rRange = ws.Range("H7:H19") Else Set rRange = ws.Range("G7:G19")
            Else
                If useT1 Then Set rRange = ws.Range("K7:K19") Else Set rRange = ws.Range("J7:J19")
            End If
        Case "USD"
            Set yRange = ws.Range("S7:S19")
            If curveType = "OIS" Then
                If useT1 Then Set rRange = ws.Range("V7:V19") Else Set rRange = ws.Range("U7:U19")
            ElseIf curveType = "GOV" Then
                If useT1 Then Set rRange = ws.Range("Y7:Y19") Else Set rRange = ws.Range("X7:X19")
            Else
                If useT1 Then Set rRange = ws.Range("AB7:AB19") Else Set rRange = ws.Range("AA7:AA19")
            End If
        Case "GBP"
            Set yRange = ws.Range("AJ7:AJ19")
            If curveType = "OIS" Then
                If useT1 Then Set rRange = ws.Range("AM7:AM19") Else Set rRange = ws.Range("AL7:AL19")
            ElseIf curveType = "GOV" Then
                If useT1 Then Set rRange = ws.Range("AP7:AP19") Else Set rRange = ws.Range("AO7:AO19")
            Else
                If useT1 Then Set rRange = ws.Range("AS7:AS19") Else Set rRange = ws.Range("AR7:AR19")
            End If
        Case Else
            InterpCurveValue = CVErr(xlErrNA): Exit Function
    End Select

    InterpCurveValue = LinInterp(yRange, rRange, years)
    Exit Function
ErrHandler:
    InterpCurveValue = CVErr(xlErrNA)
End Function

' Linear interpolation on (x=tenor years, y=rate) with flat extrapolation at ends.
Private Function LinInterp(xRange As Range, yRange As Range, x As Double) As Variant
    Dim xa() As Double, ya() As Double, n As Long, i As Long
    Dim cx As Range, cy As Range
    Dim xs As Variant, ys As Variant
    xs = xRange.Value: ys = yRange.Value

    ReDim xa(1 To UBound(xs, 1))
    ReDim ya(1 To UBound(ys, 1))
    n = 0
    For i = 1 To UBound(xs, 1)
        If IsNumeric(xs(i, 1)) And IsNumeric(ys(i, 1)) Then
            If xs(i, 1) <> "" And ys(i, 1) <> "" Then
                n = n + 1
                xa(n) = CDbl(xs(i, 1))
                ya(n) = CDbl(ys(i, 1))
            End If
        End If
    Next i
    If n = 0 Then LinInterp = CVErr(xlErrNA): Exit Function
    If n = 1 Then LinInterp = ya(1): Exit Function

    ' simple insertion sort by x (nodes are few)
    Dim j As Long, tx As Double, ty As Double
    For i = 2 To n
        tx = xa(i): ty = ya(i): j = i - 1
        Do While j >= 1
            If xa(j) <= tx Then Exit Do
            xa(j + 1) = xa(j): ya(j + 1) = ya(j): j = j - 1
        Loop
        xa(j + 1) = tx: ya(j + 1) = ty
    Next i

    If x <= xa(1) Then LinInterp = ya(1): Exit Function
    If x >= xa(n) Then LinInterp = ya(n): Exit Function
    For i = 1 To n - 1
        If x >= xa(i) And x <= xa(i + 1) Then
            If xa(i + 1) = xa(i) Then
                LinInterp = ya(i)
            Else
                LinInterp = ya(i) + (ya(i + 1) - ya(i)) * (x - xa(i)) / (xa(i + 1) - xa(i))
            End If
            Exit Function
        End If
    Next i
    LinInterp = ya(n)
End Function

Private Function OpenOPICS(conn As Object, wsCfg As Worksheet) As Boolean
    On Error GoTo Fail
    Dim serverName As String, uid As String, pwd As String, connStr As String
    serverName = Replace(CleanText(wsCfg.Range(CFG_DSN).Value), "/", "\")
    uid = CleanText(wsCfg.Range(CFG_UID).Value)

    If serverName = "" Then
        MsgBox "Missing OPICS server/DSN in Config " & CFG_DSN & ".", vbExclamation
        OpenOPICS = False: Exit Function
    End If

    If uid = "" Then
        connStr = "Provider=MSOLEDBSQL;Data Source=" & serverName & _
                  ";Initial Catalog=OpicsMain;Integrated Security=SSPI;"
    Else
        pwd = CStr(wsCfg.Range("B16").Value)   ' optional SQL password cell
        connStr = "Provider=MSOLEDBSQL;Data Source=" & serverName & _
                  ";Initial Catalog=OpicsMain;User ID=" & uid & ";Password=" & pwd & ";"
    End If

    Dim t As Long: t = val(CleanText(wsCfg.Range(CFG_BBG_TIMEOUT).Value))
    If t <= 0 Then t = 30
    conn.ConnectionTimeout = t
    conn.CommandTimeout = 120
    conn.Open connStr
    OpenOPICS = (conn.State = 1)
    Exit Function
Fail:
    MsgBox "OPICS connection failed:" & vbCrLf & Err.Description, vbCritical
    OpenOPICS = False
End Function

Private Function TrimChar(s As String, ch As String) As String
    Dim r As String: r = s
    Do While Len(r) > 0 And Left$(r, 1) = ch
        r = Mid$(r, 2)
    Loop
    Do While Len(r) > 0 And Right$(r, 1) = ch
        r = Left$(r, Len(r) - 1)
    Loop
    TrimChar = r
End Function

' Builds a SQL IN-list fragment from a comma list, e.g. "H" -> "'H'" ; "A,T" -> "'A','T'"
Private Function BuildInClause(csv As String) As String
    Dim parts() As String, i As Long, out As String, t As String
    If CleanText(csv) = "" Then BuildInClause = "": Exit Function
    parts = Split(csv, ",")
    For i = LBound(parts) To UBound(parts)
        t = Trim$(parts(i))
        If t <> "" Then
            If out <> "" Then out = out & ","
            out = out & "'" & EscapeSQL(t) & "'"
        End If
    Next i
    BuildInClause = out
End Function

Private Function AppendCsv(existing As String, item As String) As String
    If Trim$(existing) = "" Then
        AppendCsv = item
    ElseIf InStr(1, "," & existing & ",", "," & item & ",") = 0 Then
        AppendCsv = existing & "," & item
    Else
        AppendCsv = existing
    End If
End Function


' =============================================================================
' BLOOMBERG REFRESH + WAIT
'   Triggers the BBG add-in recalc on the given ranges and waits for the
'   #N/A Requesting Data... cells to clear (or timeout).
' =============================================================================

'   Compile-safe: ParamArray items are Variants, so each is cast to Range before
'   use; RangeHasPendingBBG / WaitSeconds take ByVal to avoid ByRef mismatch.
Private Sub RefreshBloombergAndWaitMany(ByVal wsCfg As Worksheet, ParamArray rngs() As Variant)
    Dim i As Long
    Dim rg As Range

    On Error Resume Next
    Application.Run "RefreshAllStaticData"     ' BBG add-in entry point (no-op if absent)
    Application.Run "RefreshCurrentSelection"
    Application.Run "BLP.RefreshAll"
    On Error GoTo 0

    Dim timeoutSec As Long
    timeoutSec = val(CleanText(wsCfg.Range(CFG_BBG_TIMEOUT).Value))
    If timeoutSec <= 0 Then timeoutSec = 60

    Dim t0 As Single: t0 = Timer
    Dim pending As Boolean
    Do
        DoEvents
        Application.Calculate
        pending = False
        For i = LBound(rngs) To UBound(rngs)
            If TypeName(rngs(i)) = "Range" Then
                Set rg = rngs(i)
                If RangeHasPendingBBG(rg) Then
                    pending = True
                    Exit For
                End If
            End If
        Next i
        If Not pending Then Exit Do
        If (Timer - t0) > timeoutSec Then Exit Do
        WaitSeconds 0.5
    Loop
End Sub

Private Function RangeHasPendingBBG(ByVal rng As Range) As Boolean
    Dim c As Range, v As Variant
    For Each c In rng.Cells
        v = c.Value
        If IsError(v) Then
            If InStr(1, c.Text, "Requesting", vbTextCompare) > 0 Then
                RangeHasPendingBBG = True
                Exit Function
            End If
        ElseIf VarType(v) = vbString Then
            If InStr(1, CStr(v), "Requesting Data", vbTextCompare) > 0 _
               Or InStr(1, CStr(v), "#N/A Requesting", vbTextCompare) > 0 Then
                RangeHasPendingBBG = True
                Exit Function
            End If
        End If
    Next c
    RangeHasPendingBBG = False
End Function

Private Sub WaitSeconds(ByVal s As Double)
    Dim t As Single: t = Timer
    Do While Timer - t < s
        DoEvents
    Loop
End Sub

' BDH writer used by T0 curve snapshot (single-point history -> "" on miss)
Private Sub WriteBDHDown(ws As Worksheet, tickerCol As String, outCol As String, _
                         firstRow As Long, lastRow As Long, fieldName As String)

    Dim n As Long
    Dim cfgT0 As String

    n = ws.Range(tickerCol & "1").Column
    cfgT0 = CfgR1C1(CFG_T0_DATE)

    ws.Range(outCol & firstRow & ":" & outCol & lastRow).FormulaR1C1 = _
        "=IF(RC" & n & "="""",""""," & _
            BDHLastPointExprR1C1( _
                "RC" & n, _
                XlText(fieldName), _
                "INT(" & cfgT0 & ")-3", _
                "INT(" & cfgT0 & ")" _
            ) & _
        ")"

End Sub

' GetAsOfDateString -> 'yyyy-mm-dd' from the T1 (as-of) config date.
Private Function GetAsOfDateString(wsCfg As Worksheet) As String
    Dim d As Variant: d = wsCfg.Range(CFG_T1_DATE).Value
    If IsDate(d) Then
        GetAsOfDateString = Format$(CDate(d), "yyyy-mm-dd")
    Else
        GetAsOfDateString = Format$(Date, "yyyy-mm-dd")
    End If
End Function

' Sets T1 = last business day, T0 = prior business day, anchored to 5pm UTC cutoff.
Private Sub Update_Config_Dates_5pmUTC(wsCfg As Worksheet)
    ' Only auto-populate if the user has not pinned dates manually.
    If CleanText(wsCfg.Range("B6").Value) = "AUTO" Then
        Dim t1 As Date, t0 As Date
        t1 = PrevBusinessDay(Date)
        t0 = PrevBusinessDay(t1)
        wsCfg.Range(CFG_T1_DATE).Value = t1
        wsCfg.Range(CFG_T0_DATE).Value = t0
    End If
End Sub

Private Function PrevBusinessDay(d As Date) As Date
    Dim x As Date: x = d - 1
    Do While Weekday(x, vbMonday) >= 6   ' Sat/Sun
        x = x - 1
    Loop
    PrevBusinessDay = x
End Function

Private Function GetOrSetT0SnapshotDateTime(wsCfg As Worksheet) As Date
    Dim v As Variant
    v = wsCfg.Range(CFG_T0_DATE).Value

    Dim dt As Date

    If IsDate(v) Then
        dt = CDate(v)

        ' If Config!B4 only has a date and no time, assume 17:00.
        If CDbl(dt) = Fix(CDbl(dt)) Then
            dt = DateValue(dt) + TimeSerial(17, 0, 0)
        End If
    Else
        ' If Config!B4 is blank/invalid:
        ' use today 17:00 if today is a weekday and now is already past 17:00,
        ' otherwise use previous business day 17:00.
        If Weekday(Date, vbMonday) <= 5 And Time >= TimeSerial(17, 0, 0) Then
            dt = Date + TimeSerial(17, 0, 0)
        Else
            dt = PrevBusinessDay(Date) + TimeSerial(17, 0, 0)
        End If
    End If

    wsCfg.Range(CFG_T0_DATE).Value = dt
    wsCfg.Range(CFG_T0_DATE).NumberFormat = "dd/mm/yyyy hh:mm"

    GetOrSetT0SnapshotDateTime = dt
End Function

' Column-signature dump for a sheet (diagnostic).  All args ByVal; first arg is a
' Workbook (never a string).  Safe call:  Call SigForSheet(ThisWorkbook, "Bonds", 4, 5)
Private Sub SigForSheet(ByVal wb As Workbook, ByVal sheetName As String, ByVal headerRow As Long, ByVal dataRow As Long)
    Dim ws As Worksheet, wsOut As Worksheet
    Dim sigHeaders As Variant
    Dim lastCol As Long, c As Long, outRow As Long

    On Error Resume Next
    Set ws = wb.Worksheets(sheetName)
    Set wsOut = wb.Worksheets(SH_DIAG)
    On Error GoTo 0

    If ws Is Nothing Then Exit Sub
    If wsOut Is Nothing Then
        SetupDiagnosticsSheet
        Set wsOut = wb.Worksheets(SH_DIAG)
    End If
    If wsOut Is Nothing Then Exit Sub

    sigHeaders = Array("Col", "Header", "Address", "Formula", "Value", _
                       "NumberFormat", "Width", "Hidden")
    wsOut.Cells(headerRow, 1).Resize(1, UBound(sigHeaders) - LBound(sigHeaders) + 1).Value = sigHeaders
    wsOut.Cells(headerRow, 1).Resize(1, UBound(sigHeaders) - LBound(sigHeaders) + 1).Font.Bold = True

    lastCol = ws.Cells(dataRow - 1, ws.Columns.Count).End(xlToLeft).Column
    If lastCol < 1 Then Exit Sub

    outRow = headerRow + 1
    For c = 1 To lastCol
        wsOut.Cells(outRow, 1).Value = c
        wsOut.Cells(outRow, 2).Value = ws.Cells(dataRow - 1, c).Value
        wsOut.Cells(outRow, 3).Value = ws.Cells(dataRow, c).Address(False, False)
        wsOut.Cells(outRow, 4).Value = ws.Cells(dataRow, c).Formula
        wsOut.Cells(outRow, 5).Value = ws.Cells(dataRow, c).Text
        wsOut.Cells(outRow, 6).Value = ws.Cells(dataRow, c).NumberFormat
        wsOut.Cells(outRow, 7).Value = ws.Columns(c).ColumnWidth
        wsOut.Cells(outRow, 8).Value = ws.Columns(c).Hidden
        outRow = outRow + 1
    Next c
End Sub

Private Sub LogDiagnosticsForRanges(ByVal stepName As String, ParamArray rngs() As Variant)
    If CleanText(ThisWorkbook.Worksheets(SH_CONFIG).Range(CFG_DIAGNOSTICS_ENABLED).Value) = "FALSE" Then Exit Sub

    SetupDiagnosticsSheet

    Dim ws As Worksheet: Set ws = ThisWorkbook.Worksheets(SH_DIAG)
    Dim nextRow As Long
    nextRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row + 1
    If nextRow < 2 Then nextRow = 2

    Dim i As Long, c As Range, rg As Range, v As Variant
    Dim etype As String, sampleText As String
    For i = LBound(rngs) To UBound(rngs)
        If TypeName(rngs(i)) = "Range" Then
            Set rg = rngs(i)
            For Each c In rg.Cells
                v = c.Value
                etype = ""
                sampleText = ""
                If IsError(v) Then
                    etype = "ERROR"
                    sampleText = c.Text
                ElseIf VarType(v) = vbString Then
                    sampleText = CStr(v)
                    If InStr(1, sampleText, "#N/A", vbTextCompare) > 0 Or _
                       InStr(1, sampleText, "Invalid", vbTextCompare) > 0 Or _
                       InStr(1, sampleText, "Not Applicable", vbTextCompare) > 0 Or _
                       InStr(1, sampleText, "Authorization", vbTextCompare) > 0 Or _
                       InStr(1, sampleText, "Requesting Data", vbTextCompare) > 0 Then
                        etype = "BBG_NA"
                    End If
                End If
                If etype <> "" Then
                    ws.Cells(nextRow, 1).Value = Now()
                    ws.Cells(nextRow, 2).Value = stepName
                    ws.Cells(nextRow, 3).Value = c.Worksheet.Name
                    ws.Cells(nextRow, 4).Value = c.Address(False, False)
                    ws.Cells(nextRow, 5).Value = etype
                    ws.Cells(nextRow, 6).Value = Left$(sampleText, 60)
                    nextRow = nextRow + 1
                End If
            Next c
        End If
    Next i
End Sub

Private Sub DumpFormulaRange(ByVal rng As Range, ByVal wsDiag As Worksheet, ByRef nextRow As Long)

    Dim c As Range
    Dim a As Range

    For Each a In rng.Areas
        For Each c In a.Cells

            If c.HasFormula Then
                wsDiag.Cells(nextRow, 1).Value = Now()
                wsDiag.Cells(nextRow, 2).Value = c.Worksheet.Name
                wsDiag.Cells(nextRow, 3).Value = c.Address(False, False)
                wsDiag.Cells(nextRow, 4).Value = c.Text
                wsDiag.Cells(nextRow, 5).Value = c.FormulaR1C1
                nextRow = nextRow + 1
            End If

        Next c
    Next a

End Sub
